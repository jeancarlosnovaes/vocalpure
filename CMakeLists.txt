cmake_minimum_required(VERSION 3.22)
project(VocalPure VERSION 0.4.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  # Keep the Windows package independent of the Visual C++ Redistributable.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/JUCE/CMakeLists.txt")
  message(FATAL_ERROR "Clone JUCE 8.0.6 into external/JUCE; see docs/build.md")
endif()
add_subdirectory(external/JUCE)
include(cmake/RNNoise.cmake)
set(formats VST3 Standalone)
if(APPLE)
  list(APPEND formats AU)
endif()
juce_add_plugin(VocalPure COMPANY_NAME "VocalPure" BUNDLE_ID "audio.vocalpure.plugin"
  PLUGIN_MANUFACTURER_CODE Vcpr PLUGIN_CODE Vp01 FORMATS ${formats}
  PRODUCT_NAME "VocalPure" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE)

if(WIN32)
  # JUCE 8.0.6 does not depend on Info.txt when generating this resource.
  # Regenerate it after a version change, including in an existing build tree.
  get_target_property(vocalpure_generated VocalPure JUCE_GENERATED_SOURCES_DIRECTORY)
  get_target_property(vocalpure_info VocalPure JUCE_INFO_FILE)
  add_custom_command(OUTPUT "${vocalpure_generated}/VocalPure_resources.rc"
    APPEND DEPENDS "${vocalpure_info}")
endif()
target_sources(VocalPure PRIVATE src/PluginProcessor.cpp src/PluginEditor.cpp src/audio/AudioEngine.cpp src/ai/RNNoiseEngine.cpp)
target_include_directories(VocalPure PUBLIC src)
target_compile_definitions(VocalPure PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(VocalPure PRIVATE juce::juce_audio_utils juce::juce_dsp
  PUBLIC vocalpure_rnnoise juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(VocalPureTests PRODUCT_NAME "VocalPure Tests")
target_sources(VocalPureTests PRIVATE tests/FoundationTests.cpp)
target_link_libraries(VocalPureTests PRIVATE VocalPure juce::juce_audio_utils juce::juce_dsp)
add_test(NAME foundation COMMAND VocalPureTests)
if(WIN32)
  add_library(VocalPureInstallCore STATIC installer/InstallCore.cpp)
  target_include_directories(VocalPureInstallCore PUBLIC installer)
  target_compile_definitions(VocalPureInstallCore PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(VocalPureInstallCore PRIVATE /W4 /utf-8)
  target_link_libraries(VocalPureInstallCore PUBLIC bcrypt ole32)
  configure_file(installer/Installer.rc.in "${CMAKE_CURRENT_BINARY_DIR}/Installer.rc" @ONLY)
  add_executable(VocalPureInstaller WIN32 installer/WindowsInstaller.cpp "${CMAKE_CURRENT_BINARY_DIR}/Installer.rc")
  target_link_libraries(VocalPureInstaller PRIVATE VocalPureInstallCore comctl32 shell32 uuid)
  target_compile_options(VocalPureInstaller PRIVATE /W4 /utf-8)
  target_link_options(VocalPureInstaller PRIVATE /MANIFEST:NO)
  set_target_properties(VocalPureInstaller PROPERTIES OUTPUT_NAME "Instalar VocalPure"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Installer/$<CONFIG>")
  add_executable(VocalPureInstallerTests tests/InstallerTests.cpp)
  target_link_libraries(VocalPureInstallerTests PRIVATE VocalPureInstallCore)
  target_compile_options(VocalPureInstallerTests PRIVATE /W4 /utf-8)
  add_test(NAME installer_transactions COMMAND VocalPureInstallerTests "${CMAKE_CURRENT_BINARY_DIR}/installer-tests")

  juce_add_console_app(VocalPureHostTests PRODUCT_NAME "VocalPure VST3 Smoke Tests")
  target_sources(VocalPureHostTests PRIVATE tests/VST3HostTests.cpp)
  target_compile_definitions(VocalPureHostTests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(VocalPureHostTests PRIVATE juce::juce_audio_processors juce::juce_recommended_config_flags)
  add_dependencies(VocalPureHostTests VocalPure_VST3)
  add_test(NAME vst3_smoke COMMAND VocalPureHostTests
    "${CMAKE_CURRENT_BINARY_DIR}/VocalPure_artefacts/$<CONFIG>/VST3/VocalPure.vst3")
endif()
